VVDEC

### Install VVDec

git clone https://github.com/fraunhoferhhi/vvdec.git
cd vvdec
make install-prefix=/usr/local -j 16
sudo make install

#### Add support to VVDec in FFMPEG

wget -O ~/libvvdec.patch https://raw.githubusercontent.com/wiki/fraunhoferhhi/vvdec/data/patch/v7-0001-avcodec-add-external-dec-libvvdec-for-H266-VVC.patch
cd <ffmpeg-path>
patch -p 1 < ~/libvvdec.patch

FFMPEG

git clone https://git.ffmpeg.org/ffmpeg.git ffmpeg
cd ffmpeg
git checkout n7.1

####
# Install the dependencies
####

./configure \
  --prefix=/usr/local \
  --bindir=/usr/local/bin \
  --enable-shared \
  --extra-cflags=-I/usr/local/include \
  --enable-gpl \
  --enable-gnutls \
  --enable-libsvtav1 \
  --enable-libdav1d \
  --enable-libsvtjpegxs \
  --enable-libvpx \
  --enable-libx264 \
  --enable-libx265 \
  --enable-libvvenc \
  --enable-libvvdec

make -j $(nproc)
sudo make install
